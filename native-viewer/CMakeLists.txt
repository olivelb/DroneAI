cmake_minimum_required(VERSION 3.24)
project(GSTileViewer VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Build with the Windows x64 MSVC toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
find_program(FXC_EXECUTABLE fxc REQUIRED)
file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/shaders.hlsl" SHADER_SOURCE_NATIVE)
set(SHADER_HEADERS)
foreach(entry Project Histogram Prefix Scatter Pick VS PS)
  set(profile cs_5_0)
  set(compute_stage 1)
  if(entry STREQUAL "VS")
    set(profile vs_5_0)
    set(compute_stage 0)
  elseif(entry STREQUAL "PS")
    set(profile ps_5_0)
    set(compute_stage 0)
  endif()
  set(header "${CMAKE_CURRENT_BINARY_DIR}/generated/${entry}.hpp")
  add_custom_command(OUTPUT "${header}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/generated"
    COMMAND "${FXC_EXECUTABLE}" /nologo /D COMPUTE_STAGE=${compute_stage} /T ${profile} /E ${entry} /O3 /Fh "${header}" /Vn shader${entry} "${SHADER_SOURCE_NATIVE}"
    DEPENDS shaders.hlsl VERBATIM)
  list(APPEND SHADER_HEADERS "${header}")
endforeach()
add_library(gstile_core src/bundle.cpp)
target_include_directories(gstile_core PUBLIC src)
target_link_libraries(gstile_core PUBLIC nlohmann_json::nlohmann_json bcrypt)
target_compile_definitions(gstile_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(GSTileViewer WIN32 src/main.cpp src/renderer.cpp ${SHADER_HEADERS})
target_include_directories(GSTileViewer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(GSTileViewer PRIVATE gstile_core d3d11 dxgi comctl32 shell32 ole32 windowscodecs)
add_executable(gstile_tests tests/tests.cpp)
target_link_libraries(gstile_tests PRIVATE gstile_core)
foreach(target gstile_core GSTileViewer gstile_tests)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()
enable_testing()
add_test(NAME gstile_contracts COMMAND gstile_tests)
install(TARGETS GSTileViewer RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)

install(FILES QUALIFICATION.md ../LICENSE DESTINATION .)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION . RENAME LICENSE-nlohmann-json.txt)

install(FILES ../docs/benchmarks/gstile-progressive-streaming-qualification.md
  DESTINATION . RENAME STREAMING-QUALIFICATION.md)
install(FILES ../docs/benchmarks/gstile-halo-opacity-qualification.md
  DESTINATION . RENAME CACHE-TRANSPARENCY-QUALIFICATION.md)
install(FILES ../docs/benchmarks/gstile-edge-opacity-qualification.md
  DESTINATION . RENAME EDGE-OPACITY-QUALIFICATION.md)
